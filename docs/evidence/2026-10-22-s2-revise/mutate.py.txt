env = {k: v for k, v in os.environ.items() if not k.startswith(("AIFY_AGENT", "HERDR", "AIFY_HERDR"))}
with open(OUT, "w", encoding="utf-8") as out:
    out.write(f"# S2 review-round mutants, {datetime.datetime.utcnow().isoformat()}Z\n")
    for name, repo, rel, old, new, (cwd, cmd) in ARMS:
        path = os.path.join(repo, rel)
        keep = path + ".mutant-keep"
        shutil.copyfile(path, keep)
        text = open(path, "rb").read().decode()
        if text.count(old) != 1:
            out.write(f"ARM {name}: NOT APPLIED (pattern count {text.count(old)})\n"); out.flush()
            os.remove(keep); continue
        open(path, "wb").write(text.replace(old, new).encode())
        r = subprocess.run(cmd, cwd=cwd, env=env, capture_output=True, text=True, encoding="utf-8", errors="replace")
        shutil.copyfile(keep, path)
        same = filecmp.cmp(keep, path, shallow=False)
        os.remove(keep)
        fails = [l.strip() for l in (r.stdout + r.stderr).splitlines() if l.startswith("not ok")]
        verdict = "KILLED" if r.returncode != 0 and fails else "SURVIVED"
        out.write(f"ARM {name} [{rel}]: exit {r.returncode}, {verdict}; restored byte-equal {same}\n")
        for f in fails[:4]:
            out.write(f"    {f}\n")
        out.flush()
